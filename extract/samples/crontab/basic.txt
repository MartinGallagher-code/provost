# Crontab for application maintenance
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/sbin:/bin:/usr/sbin:/usr/bin
MAILTO=admin@example.com

# Backup database every day at 2 AM
0 2 * * * /usr/local/bin/backup-db.sh

# Cleanup temp files every hour
0 * * * * find /tmp -mtime +7 -delete

# Weekly report on Sundays at midnight
0 0 * * 0 /usr/local/bin/weekly-report.sh

# Monthly audit on the 1st at 3 AM
0 3 1 * * /usr/local/bin/audit.sh --full

@reboot /usr/local/bin/startup.sh
