services:
  web:
    image: nginx:1.23
    ports:
      - "80:80"
      - "443:443"
    depends_on:
      - api
    volumes:
      - type: bind
        source: ./conf
        target: /etc/nginx
  api:
    image: myapp:2.1
    environment:
      DATABASE_URL: postgres://db/app
      LOG_LEVEL: info
    depends_on:
      - db
  db:
    image: postgres:15
    volumes:
      - pgdata:/var/lib/postgresql/data
